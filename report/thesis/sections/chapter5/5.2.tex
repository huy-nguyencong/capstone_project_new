\section{Thiết kế các luồng xử lý chính}
\label{sec:5.2}

\subsection{Luồng xử lý dữ liệu camera}
\label{sec:5.2.1}

Mọi hoạt động phân tích là các \emph{công việc xử lý} trong hàng đợi, mỗi công việc gắn với một camera và một nguồn. Với \textbf{phiên RTSP tự động}, khi hàng đợi trống, bộ lập lịch của tiến trình nền chọn camera đang vận hành, bật xử lý AI, có RTSP và chờ lâu nhất, rồi tạo một phiên 1.800 khung hình nguồn (khoảng 30 giây ở 60 khung hình mỗi giây); camera có phiên gần nhất thất bại được bỏ qua 5 phút. Các camera vì vậy được phân tích lần lượt, với những khoảng không được phân tích giữa hai phiên của cùng một camera (Mục~\ref{sec:1.4}). Với \textbf{tệp video tải lên}, máy chủ kiểm tra, lưu tạm tệp rồi tạo công việc.

\paragraph{Vòng đời công việc.} Hình~\ref{fig:job-state} mô tả các trạng thái của một công việc. Công việc \emph{Chờ xử lý} khi được tiến trình nền nhận chuyển sang \emph{Đang xử lý} với thời hạn giữ chỗ 60 giây, được gia hạn trong lúc xử lý; nếu tiến trình dừng đột ngột và hết hạn, công việc được nhận lại. Với công việc xử lý tệp video, lỗi tạm thời được thử lại sau một khoảng chờ, tối đa ba lần; phiên RTSP gặp lỗi thì kết thúc ngay và camera được xử lý lại ở một phiên sau. Công việc kết thúc ở \emph{Hoàn thành}; \emph{Thất bại} khi gặp lỗi không phục hồi được hoặc hết lượt thử; hoặc \emph{Đã hủy} khi Quản trị viên hủy, khi camera bị loại khỏi vận hành hay bị tắt xử lý AI.

\paragraph{Các bước xử lý.} Hình~\ref{fig:pipeline-flow} trình bày hai giai đoạn của một công việc. Trong \emph{vòng lặp theo khung hình}, mỗi khung hình được đọc kèm chỉ số và mốc thời gian (Mục~\ref{sec:3.7.4}), và một trong mỗi $N$ khung hình đi qua Detector và Tracker. Kết quả được cập nhật vào \emph{bộ đệm track}, nơi giữ các lần xuất hiện đang theo vết và ứng viên khung hình đại diện; khi một lần xuất hiện kết thúc, khung hình đại diện được chọn và giữ trong bộ nhớ. Khi hết nguồn hoặc hết phiên, các lần xuất hiện còn lại được kết thúc và công việc chuyển sang giai đoạn \emph{mã hóa và ghi}: với mỗi lần xuất hiện, vùng người được cắt tạm trong bộ nhớ, nới mỗi phía 5\% khung bao, mã hóa thành vector đặc trưng và ghi vào ba kho (Mục~\ref{sec:5.3.4}). Vì vậy lần xuất hiện chỉ tìm được sau khi công việc kết thúc; với RTSP, dữ liệu của camera được ghi theo từng phiên 1.800 khung hình. Công việc bị hủy hoặc hết hạn khi đang phân tích thì không ghi gì. Mỗi lần xuất hiện lưu kèm cấu hình Detector-Tracker, phiên bản bộ mã hóa và khoảng lấy mẫu $N$ đã dùng.

\subsection{Chọn khung hình đại diện}
\label{sec:5.2.2}

Mỗi lần xuất hiện chỉ lưu \emph{một} khung hình toàn cảnh, dùng cho cả vector lẫn hiển thị, nên khung hình này cần cho thấy người rõ nhất có thể. Bộ chọn hoạt động như sau:
\begin{itemize}
    \item \textbf{Giữ tối đa ba ứng viên} cho mỗi lần xuất hiện đang diễn ra, tổng bộ nhớ không quá 256~MB.
    \item \textbf{Lọc cứng.} Ứng viên cần khung bao tối thiểu 24$\times$48 điểm ảnh, vùng người đủ nét, chạm ít hơn hai cạnh khung hình và không đổi hình dạng đột ngột so với lần trước (diện tích trong 4 lần, tỉ lệ rộng-cao trong 2,5 lần), vì thay đổi đột ngột thường do nhầm người hoặc bị che khuất.
    \item \textbf{Điểm chất lượng} gồm độ lớn của người (0,30), độ nét (0,25), độ đầy đủ của thân người (0,20), độ ổn định (0,15) và độ gần giữa khoảng xuất hiện (0,10), trừ điểm phạt khi chạm cạnh khung hình hoặc chồng lấn người khác.
    \item \textbf{Chọn khi kết thúc.} Chọn ứng viên tốt nhất trong số các ứng viên đạt lọc cứng; nếu không có, vẫn giữ ứng viên tốt nhất và đánh dấu lần xuất hiện chất lượng thấp thay vì loại bỏ.
\end{itemize}
Tiêu chí và trọng số được chọn theo kinh nghiệm, lưu trong tệp cấu hình và cho kết quả xác định.

\subsection{Luồng tìm kiếm}
\label{sec:5.2.3}

Hình~\ref{fig:search-sequence} trình bày luồng tìm kiếm.

\begin{figure}[!htbp]
\centering
\includegraphics[width=\textwidth]{H4-tuan-tu-tim-kiem.png}
\caption{Sơ đồ tuần tự của luồng tìm kiếm}
\label{fig:search-sequence}
\end{figure}

Ứng dụng web gửi truy vấn cùng bộ lọc camera, khoảng thời gian và $k$. Máy chủ lần lượt:
\begin{enumerate}
    \item \textbf{kiểm tra đầu vào:} ảnh JPEG hoặc PNG tối đa 10~MB; thuộc tính thuộc danh sách cho phép và tạo thành tổ hợp hợp lệ; $k$ thuộc tập $\{4, 8, 12, 16\}$;
    \item \textbf{mã hóa truy vấn:} ảnh qua bộ mã hóa ảnh, câu mô tả qua bộ mã hóa văn bản; thuộc tính được ghép thành câu tiếng Anh theo mẫu cố định, ví dụ \emph{``A woman wearing a red jacket and blue jeans, carrying a handbag.''};
    \item \textbf{xác định phạm vi:} khu vực lấy từ tài khoản của Giám sát viên, camera là các camera được chọn hoặc mọi camera đang vận hành của khu vực; camera khác bị từ chối;
    \item \textbf{tìm kiếm vector có lọc trước} trong Milvus, đối chiếu với PostgreSQL và tìm lại gấp đôi khi thiếu kết quả hợp lệ, tối đa ba lượt (Mục~\ref{sec:3.6.4}).
\end{enumerate}
Danh sách chỉ chứa thông tin mô tả và điểm phù hợp; mỗi ảnh là một yêu cầu riêng, được kiểm tra quyền lại và dựng từ khung hình đại diện (Mục~\ref{sec:6.3.3}), nên danh sách trả về nhanh và không ảnh nào bỏ qua bước kiểm tra quyền.

\subsection{Luồng quản lý hồ sơ vụ việc}
\label{sec:5.2.4}

Hình~\ref{fig:case-sequence} trình bày luồng lưu một kết quả vào vụ việc mới hoặc đã có.

\begin{figure}[!htbp]
\centering
\includegraphics[width=0.95\textwidth]{H5-tuan-tu-vu-viec.png}
\caption{Sơ đồ tuần tự của luồng lưu kết quả vào vụ việc}
\label{fig:case-sequence}
\end{figure}

Giao diện chỉ gửi mã lần xuất hiện. Máy chủ kiểm tra lần xuất hiện đã sẵn sàng, thuộc camera đang vận hành trong khu vực hiện tại của Giám sát viên, và vụ việc do chính người đó phụ trách và đang xử lý, rồi tạo một mục mới kèm \emph{bản chụp} tên camera, khu vực và thời điểm, để vụ việc vẫn đúng khi camera đổi tên hoặc khung hình tạm thời không truy xuất được. Yêu cầu cập nhật kèm số phiên bản của vụ việc, nên thay đổi đồng thời không bị ghi đè âm thầm. Để đánh dấu hoàn thành khi lưu, giao diện đọc lại số phiên bản mới nhất sau khi lưu rồi gửi yêu cầu đổi trạng thái; nếu bước này thất bại, kết quả vẫn được giữ và người dùng được nhắc đánh dấu lại. Ảnh trong vụ việc được cấp theo quyền đối với vụ việc (Mục~\ref{sec:4.6}).
