\section{Đặc tả use case}
\label{sec:4.5}

Mục này đặc tả tám use case của luồng nghiệp vụ chính: người dùng đăng nhập (UC-01); Quản trị viên thêm camera (UC-03), đưa camera vào phân tích (UC-04) và chọn mô hình (UC-05); Giám sát viên tìm kiếm (UC-09), đánh giá kết quả (UC-10) và lưu một kết quả vào vụ việc (UC-11); Quản lý xem các vụ việc (UC-13). Ngoại lệ ký hiệu E, luồng thay thế ký hiệu A. Các use case còn lại đã được nêu trong các yêu cầu ở Mục~\ref{sec:4.2}.

\textbf{Luồng chính} & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{1.}Người dùng mở trang đăng nhập, nhập tên đăng nhập và mật khẩu. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{2.}Hệ thống kiểm tra thông tin đăng nhập và trạng thái của tài khoản. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{3.}Hệ thống tạo phiên đăng nhập và xác định vai trò của tài khoản. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{4.}Hệ thống mở trang chủ của vai trò, chỉ hiển thị chức năng của vai trò đó. \\ \hline
\textbf{Ngoại lệ} & \textbf{E1 - Sai thông tin đăng nhập:} Sai tên đăng nhập hay sai mật khẩu đều nhận cùng một thông báo, và lần thất bại được ghi nhật ký. \\ \cline{2-2}
 & \textbf{E2 - Tài khoản bị khóa hoặc ngừng hoạt động:} Hệ thống từ chối đăng nhập. \\ \cline{2-2}
 & \textbf{E3 - Thử đăng nhập quá nhiều lần:} Hệ thống tạm từ chối và cho biết thời gian cần chờ. \\ \hline
\textbf{Luồng thay thế} & Không có. \\ \hline

\begin{longtable}{|>{\raggedright\arraybackslash}p{2.9cm}|p{12.3cm}|}
\caption{Đặc tả use case UC-03: Quản lý camera}\label{tab:uc03}\\
\hline
\textbf{Tác nhân} & Quản trị viên \\ \hline
\textbf{Mục đích} & Thêm, cập nhật camera, kiểm tra kết nối RTSP, loại khỏi vận hành và đưa trở lại. \\ \hline
\textbf{Tiền điều kiện} & Quản trị viên đã đăng nhập; danh mục khu vực giám sát đã được khai báo. \\ \hline
\textbf{Hậu điều kiện} & Thông tin, trạng thái và kết quả kiểm tra RTSP của camera được lưu. \\ \hline
\endfirsthead
\multicolumn{2}{l}{\textit{(tiếp theo Bảng~\ref{tab:uc03})}}\\
\hline
\endhead
\textbf{Luồng chính} & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{1.}Quản trị viên mở quản lý camera; hệ thống liệt kê camera cùng khu vực, trạng thái vận hành, RTSP và xử lý AI. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{2.}Quản trị viên thêm camera với mã, tên, khu vực và địa chỉ RTSP, có thể kèm thông tin xác thực; camera chỉ xử lý bằng tệp video không cần địa chỉ. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{3.}Hệ thống kiểm tra dữ liệu, lưu thông tin xác thực dạng mã hóa, tách khỏi địa chỉ, và lưu camera ở trạng thái \emph{Chưa xác minh}. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{4.}Hệ thống kết nối tới luồng RTSP và cập nhật trạng thái kết nối. \\ \hline
\textbf{Ngoại lệ} & \textbf{E1 - Dữ liệu không hợp lệ:} Thiếu thông tin, địa chỉ RTSP không hợp lệ hoặc mã camera đã tồn tại thì bị từ chối. \\ \cline{2-2}
 & \textbf{E2 - Không kết nối được:} Camera được lưu ở trạng thái \emph{Mất kết nối} kèm cảnh báo, để sửa và kiểm tra lại. \\ \cline{2-2}
 & \textbf{E3 - Địa chỉ ngoài dải mạng cho phép:} Địa chỉ ngoài dải mạng camera khai báo khi triển khai không được kiểm tra; camera giữ trạng thái \emph{Chưa xác minh}. \\ \hline
\textbf{Luồng thay thế} & \textbf{A1 - Cập nhật camera:} Sửa được tên hoặc địa chỉ RTSP, không sửa được khu vực; địa chỉ mới được kiểm tra lại từ \emph{Chưa xác minh}. \\ \cline{2-2}
 & \textbf{A2 - Kiểm tra lại kết nối:} Quản trị viên yêu cầu kiểm tra kết nối lại. \\ \cline{2-2}
 & \textbf{A3 - Loại khỏi vận hành:} Xử lý AI bị tắt và công việc đang chạy bị hủy; dữ liệu được giữ nhưng không tìm kiếm được. \\ \cline{2-2}
 & \textbf{A4 - Đưa vận hành trở lại:} Camera trở lại với xử lý AI tắt, và dữ liệu tìm kiếm được trở lại. \\ \hline
\end{longtable}

\begin{longtable}{|>{\raggedright\arraybackslash}p{2.9cm}|p{12.3cm}|}
\caption{Đặc tả use case UC-04: Quản lý xử lý AI trên camera}\label{tab:uc04}\\
\hline
\textbf{Tác nhân} & Quản trị viên \\ \hline
\textbf{Mục đích} & Chọn camera nào được phân tích, và xử lý tệp video ghi sẵn cho một camera khi cần. \\ \hline
\textbf{Tiền điều kiện} & Quản trị viên đã đăng nhập; camera đã có và đang vận hành. \\ \hline
\textbf{Hậu điều kiện} & Trạng thái xử lý AI được cập nhật, và camera vào hoặc ra khỏi quá trình phân tích. \\ \hline
\endfirsthead
\multicolumn{2}{l}{\textit{(tiếp theo Bảng~\ref{tab:uc04})}}\\
\hline
\endhead
\textbf{Luồng chính} & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{1.}Quản trị viên mở quản lý xử lý AI; hệ thống liệt kê camera cùng trạng thái xử lý AI và cấu hình mô hình đang dùng. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{2.}Quản trị viên bật hoặc tắt xử lý AI cho một camera và xác nhận. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{3.}Hệ thống kiểm tra trạng thái của camera và cấu hình mô hình hiện hành. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{4.}Hệ thống lưu trạng thái. Camera được bật sẽ được phân tích lần lượt cùng các camera khác; camera bị tắt thì công việc đang chạy bị hủy và không tạo dữ liệu mới, dữ liệu cũ được giữ. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{5.}Hệ thống thông báo kết quả cho Quản trị viên. \\ \hline
\textbf{Ngoại lệ} & \textbf{E1 - Camera mất kết nối:} Không bật được xử lý AI cho camera RTSP đang mất kết nối. \\ \cline{2-2}
 & \textbf{E2 - Mô hình AI chưa sẵn sàng:} Không bật được xử lý AI khi cấu hình mô hình chưa đầy đủ hoặc không khả dụng. \\ \cline{2-2}
 & \textbf{E3 - Camera không còn vận hành:} Không đổi được trạng thái xử lý AI của camera đã ngừng vận hành. \\ \cline{2-2}
 & \textbf{E4 - Tệp video không hợp lệ:} Tệp sai định dạng hoặc dung lượng bị từ chối, không tạo công việc. \\ \hline
\textbf{Luồng thay thế} & \textbf{A1 - Xử lý tệp video:} Quản trị viên tải lên video ghi sẵn cho một camera, kèm thời điểm bắt đầu và chế độ lấy mẫu. Hệ thống kiểm tra tệp và tạo công việc dùng cùng quy trình như luồng RTSP, hiển thị tiến độ. Đây là cách duy nhất để xử lý camera không có RTSP. \\ \hline
\end{longtable}

 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{2.}Giám sát viên có thể chọn camera của khu vực, khoảng thời gian và $k$ trong 4, 8, 12, 16, rồi gửi yêu cầu. \\

 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{5.}Hệ thống hiển thị ảnh người, camera, khu vực, thời điểm và điểm phù hợp của từng kết quả. \\ \hline

 & \textbf{E2 - Không có dữ liệu trong phạm vi:} Hệ thống gợi ý mở rộng khoảng thời gian hoặc chọn thêm camera. \\ \cline{2-2}
 & \textbf{E3 - Thành phần tìm kiếm không khả dụng:} Nếu bộ mã hóa hoặc kho vector không hoạt động, tìm kiếm báo lỗi thay vì trả danh sách rỗng. \\ \cline{2-2}
 & \textbf{E4 - Yêu cầu ngoài phạm vi:} Yêu cầu gửi trực tiếp cho camera ngoài khu vực bị từ chối. \\ \hline
\textbf{Luồng thay thế} & \textbf{A1 - Tìm tiếp bằng một kết quả:} Giám sát viên kéo ảnh một kết quả vào ô tìm kiếm để tìm tiếp người đó. \\ \hline
\end{longtable}

\textbf{Tiền điều kiện} & Giám sát viên đã đăng nhập; kết quả cần lưu thuộc phạm vi tìm kiếm của Giám sát viên. \\ \hline
\textbf{Hậu điều kiện} & Vụ việc được tạo hoặc cập nhật, có một người phụ trách, tiêu đề, ghi chú, trạng thái và kết quả đã lưu. \\ \hline

\textbf{Luồng chính} & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{1.}Từ một kết quả tìm kiếm, Giám sát viên chọn \textbf{Tạo vụ việc mới} hoặc \textbf{Thêm vào vụ việc đã có}. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{2.}Vụ việc mới cần tiêu đề và ghi chú, người phụ trách là tài khoản đang đăng nhập; với vụ việc đã có, chỉ hiện các vụ việc đang xử lý của chính Giám sát viên. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{3.}Kết quả được lưu thành một mục mới, kể cả khi lần xuất hiện đã có trong vụ việc. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{4.}Giám sát viên có thể sửa tiêu đề, ghi chú và loại kết quả; lần xuất hiện gốc vẫn được giữ. \\ 
 & \hangindent=1.6em\hangafter=1\noindent\makebox[1.6em][l]{5.}Khi xong, Giám sát viên đánh dấu \textbf{Hoàn thành}; hệ thống ghi thời điểm và khóa vụ việc. \\ \hline
\textbf{Ngoại lệ} & \textbf{E1 - Kết quả hoặc vụ việc không còn khả dụng:} Không lưu gì và tải lại danh sách vụ việc. \\ \cline{2-2}
 & \textbf{E2 - Kết quả ngoài quyền truy cập:} Yêu cầu gửi trực tiếp để lưu kết quả ngoài phạm vi bị từ chối. \\ \cline{2-2}
 & \textbf{E3 - Vụ việc đã hoàn thành:} Không thay đổi được cho tới khi vụ việc được mở lại. \\ \cline{2-2}
 & \textbf{E4 - Dữ liệu đã bị thay đổi ở nơi khác:} Cập nhật bị từ chối và phải tải lại vụ việc. \\ \hline
\textbf{Luồng thay thế} & \textbf{A1 - Đánh dấu hoàn thành ngay khi lưu:} Khi bật \textbf{Đánh dấu vụ việc đã hoàn thành}, vụ việc được hoàn thành ngay sau khi lưu; nếu bước này thất bại, kết quả vẫn được lưu và Giám sát viên được nhắc đánh dấu lại. \\ \cline{2-2}
 & \textbf{A2 - Mở lại vụ việc:} Vụ việc đã hoàn thành trở về \textbf{Đang xử lý} và sửa được. \\ \hline
\end{longtable}
